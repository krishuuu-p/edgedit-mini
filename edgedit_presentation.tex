\begin{frame}{What is a surrogate surgery?}
\small
The object being changed is a \textbf{teacher transformer} made of six full blocks.

\vspace{0.8em}
\begin{center}
  \textbf{Teacher:}\quad [B1] \ [B2] \ [B3] \ [B4] \ [B5] \ [B6]\\[0.4em]
  \textit{\footnotesize Each block: attention dimension $d=128$, MLP ratio $r=4$}
\end{center}

\vspace{0.8em}
\textbf{Possible distilled substitutions:}
\begin{itemize}
  \itemsep0.2em
  \item \textbf{Block removal:}\hspace{4.2em} [B1 + B2] $\rightarrow$ [M12]
  \item \textbf{MLP reduction:}\hspace{3.8em} [Bi, $r=4$] $\rightarrow$ [Bi, $r=2$]
  \item \textbf{Hidden-dim reduction:}\quad [Bi, $d=128$] $\rightarrow$ [Bi, $d=64$]
\end{itemize}

\vspace{0.8em}
Each substitute is trained with FwKD to match the corresponding teacher feature.

\vspace{0.5em}
\begin{block}{}
  \centering
  A candidate EdgeDiT is formed by retaining some teacher blocks and replacing others with these distilled surrogates.
\end{block}
\end{frame}

\begin{frame}{Hardware-Aware Search and Main Results}
\centering
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\small\textbf{Reduced-scale setup}
\begin{itemize}\scriptsize
  \setlength{\itemsep}{0pt}
  \item FashionMNIST $28\times28$; Tiny-DiT teacher: 1.875M params
  \item Teacher: 30K steps; 60 candidate architectures
  \item Surrogates: 1.6K steps each; selected models: 15K fine-tuning steps
  \item Measured RTX 3050 latency + feature MSE for Pareto selection
\end{itemize}

\vspace{0.3em}
{\centering
  \includegraphics[width=\linewidth,height=4.0cm,keepaspectratio]{pareto_front.png}\par

}

\column{0.50\textwidth}
\footnotesize

 & \textbf{(M)} & \textbf{(ms/step)} & \textbf{$\downarrow$} \\

\begin{block}{EdgeDiT-small: maximum efficiency}
\centering\footnotesize
\alert{47.5\% fewer parameters} \quad \alert{2.27$\times$ faster} per step\\[0.2em]
\scriptsize Achieves aggressive compression with a moderate quality trade-off.
\end{block}

\begin{block}{EdgeDiT-large: best quality trade-off}
\centering\footnotesize
\alert{39.6\% lower Proxy-FID} \quad \alert{1.54$\times$ faster} than teacher\\[0.2em]
\scriptsize Improves generation quality using fewer parameters and lower latency.
\end{block}
\end{columns}

\begin{frame}{Conclusion}
\begin{block}{What was reproduced successfully}
\begin{itemize}
  \item All three EdgeDiT surgery operations: block removal, MLP-ratio reduction, and hidden-dimension reduction.
  \item Feature-wise surrogate distillation, candidate assembly, latency-aware Pareto search, and end-to-end fine-tuning.
  \item The paper's core conclusion: compact diffusion transformers can achieve strong quality-efficiency trade-offs through distillation and hardware-aware selection.
\end{itemize}
\end{block}

\begin{block}{Final takeaway}
\centering
\alert{EdgeDiT-small} demonstrates substantial compression and speed; \alert{EdgeDiT-large} achieves the best generation quality while still being faster than the teacher.
\end{block}
